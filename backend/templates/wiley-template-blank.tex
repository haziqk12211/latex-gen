%%
%% Blank Wiley (WileyNJDv5) article skeleton
%%
%% Documentclass options select citation style and column layout:
%%   HARVARD / VANCOUVER / NUMBERED / AMA  -> citation/reference style
%%   LATO2COL (two-column) or LATO1COL (one-column) -> layout
%% Swap these per the target journal's author guidelines.
\documentclass[HARVARD,LATO2COL]{WileyNJDv5}

%% ---------------------------------------------------------------
%% Journal/article metadata — filled in by the editorial office or
%% left as placeholders pre-submission.
%% ---------------------------------------------------------------
\articletype{}%
\received{}
\revised{}
\accepted{}
\journal{}
\volume{}
\copyyear{}
\startpage{}

\raggedbottom

\begin{document}

\title{}

%% One \author[n]{} per author; the [n] links to the matching
%% \address[n]{} below for shared/multiple affiliations.
\author[1]{}
\author[2]{}

%% Running header short author list, e.g. "Smith et al."
\authormark{}
%% Running header short title
\titlemark{}

%% \address[n]{} — repeat per distinct affiliation, matching the
%% \author[n] numbering above.
\address[1]{\orgdiv{}, \orgname{}, \orgaddress{\state{}, \country{}}}

%% Corresponding author note + contact email.
\corres{\email{}}

%% Optional short-title argument in [ ] for running heads/TOC.
\abstract[]{}

\keywords{}

\maketitle

\section{Introduction}

\section{Methodology}

\section{Results}

\section{Discussion}

\section{Conclusion}

%% Figures, tables, and equations follow standard LaTeX conventions
%% (graphicx figure/table environments, equation/align environments) —
%% WileyNJDv5 does not require special caption placement rules beyond
%% those built into the class.

%% Bibliography file name should match the citation style set above
%% (e.g. wileyNJD-AMA, wileyNJD-Harvard, wileyNJD-Vancouver).
\bibliography{}

\end{document}
