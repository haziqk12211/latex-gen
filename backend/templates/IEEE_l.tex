%% IEEEtran journal article skeleton, trimmed to fields this app collects.
\documentclass[journal]{IEEEtran}

%% FormData fields that are close to IEEE standard options, use standards instead of specific FormData

%% Keep amsmath/cite - commonly needed regardless of content.
\usepackage{amsmath}
\usepackage{cite}

\begin{document}

%% Title -> FormData.title
\title{}

%% One \author block per entry in FormData.authors. Use ~ between name
%% parts so names don't break across lines. \thanks{} holds affiliation
%% text (FormData.authors[i].affiliation) and email - one \thanks{} per author.
%% add ',~' at the end of each author's name in \author{} except for the final author's name.
%% ORCID (FormData.authors[i].orcidId) is not a native IEEEtran field;
%% append it inside the same \thanks{} block, e.g. "ORCID: 0000-...", if provided
%% mention if they are a correspong author at the end of the \thanks{} block in brackets :().
%% make sure BOTH \author{} and \thanks{} blocks exist. 


\author{
  ~\IEEEmembership{}%
\thanks{}
}

\maketitle

%% if highlights are required, add a section and some placeholders


%% Abstract -> FormData.abstract
\begin{abstract}

\end{abstract}

%% Keywords -> FormData.keywords, joined using FormData.keywordSeparator
\begin{IEEEkeywords}

\end{IEEEkeywords}

%% Required boilerplate call - keep regardless of review mode.
\IEEEpeerreviewmaketitle

%% One \section{} per entry the user checked in FormData.keySections
%% (Introduction, Literature Review, Methodology, Experimentation,
%% Results, Discussion). Generate only the sections that were checked -
%% do not invent additional ones, also add one word place holders
%% add an empty placeholder table in any one key section to show table font,
%% for the table use [htbp]
\section{Introduction}

\section{Conclusion}

%% Declarations/statements - only include a subsection here if the , uses \section*
%% corresponding FormData field is non-empty:
%%   FormData.fundingStatement        -> "Funding" statement
%%   FormData.conflictOfInterest      -> "Conflict of Interest" statement
%%   FormData.dataAvailability        -> "Data Availability" statement
%%   FormData.ethicsApproval          -> "Ethics Approval" statement
%%   FormData.consentForPublication   -> "Consent for Publication" statement
%%   FormData.authorContributions    -> "Author Contributions" statement
%%   FormData.creditStatement         -> CRediT statement (Elsevier only)
%%   FormData.generativeAI            -> Generative AI use disclosure
%% Acknowledgment uses \section* (not \section):
\section*{Acknowledgment}

%% Bibliography style -> FormData.referencingStyle.
%% If FormData.bibliographyText or an uploaded .bib file is present,
